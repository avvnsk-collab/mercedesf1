zpm build

项目打包

Options:
  -h, --help             Show help                                     [boolean]
  -v, --version          Show version number                           [boolean]
  -t, --target           Target to build                           [default: ""]
  -p, --path             Path to project                          [default: "."]
      --png2vg           image conversion              [boolean] [default: true]
      --jsc              Compile JS file to C or bin, [--jsc c|bin][default: ""]
      --prune            Prune unused js files        [boolean] [default: false]
      --pack             Pack zpk and zab or not       [boolean] [default: true]
      --show-info        Show all image convert and js compile information lines
                                                      [boolean] [default: false]
  -c, --compress         Compress or not                        [default: false]
      --ip               Include Intermediate Products package
                                                       [boolean] [default: true]
      --legacy-zpk-name  Use legacy target-based zpk filename instead of md5
                         hash                         [boolean] [default: false]
      --px               Px or not                               [default: true]
  -m, --metrics          Do Metrics Parse process                [default: true]
      --sources          device info                       [array] [default: []]
      --st               target platform st info           [array] [default: []]
      --sr               target platform sr info           [array] [default: []]
      --cp               target platform cp info           [array] [default: []]
      --dw               target platform dw info           [array] [default: []]
